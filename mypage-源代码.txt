  <script>
    // 1. find the elements
    const heading = document.getElementById('myHeading');
    const button = document.getElementById('myButton');
    const resetButton = document.getElementById('resetButton');
    const clickCount = document.getElementById('clickCount');
    const colorName = document.getElementById('colorName');
    const colorSwatch = document.getElementById('colorSwatch');
    const stateText = document.getElementById('stateText');

    // 2. fun messages and colors
    const funMessages = [
      "You're awesome! 🚀", "JavaScript is magic! ✨", "Great job! 🎉",
      "Clicking feels good! 😄", "You found the fun! 🎈", "So interactive! 🕹️",
      "Nice click! 💥", "High five! 🙌", "Welcome to the future! 🤖"
    ];
    const colors = [
      { name: "coral red",        value: "#FF6B6B" },
      { name: "sunshine yellow",  value: "#FBBF24" },
      { name: "emerald green",    value: "#34D399" },
      { name: "sky blue",         value: "#60A5FA" },
      { name: "bubblegum pink",   value: "#F472B6" },
      { name: "lavender purple",  value: "#A78BFA" },
      { name: "tangerine orange", value: "#F97316" },
      { name: "teal",             value: "#2DD4BF" }
    ];

    // 3. click the button → new text + new color + count
    let clicks = 0;
    button.addEventListener('click', function (e) {
      clicks++;
      clickCount.textContent = clicks;
      const msg = funMessages[Math.floor(Math.random() * funMessages.length)];
      const c = colors[Math.floor(Math.random() * colors.length)];
      heading.textContent = msg;
      heading.style.color = c.value;
      colorName.textContent = c.name;
      colorSwatch.style.background = c.value;
      stateText.textContent = 'fired!';
      spawnSparks(e.clientX, e.clientY);
      heading.classList.add('pop');
      setTimeout(function () { heading.classList.remove('pop'); }, 400);
    });

    // 4. reset button → back to the beginning
    resetButton.addEventListener('click', function () {
      clicks = 0;
      clickCount.textContent = 0;
      heading.textContent = 'Welcome! 👋';
      heading.style.color = '#F8FAFC';
      colorName.textContent = 'snow white';
      colorSwatch.style.background = '#F8FAFC';
      stateText.textContent = 'reset';
    });

    // 5. little sparkles that fly out when you click
    function spawnSparks(x, y) {
      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
      const sparkEmojis = ['🎉', '✨', '💥', '⭐'];
      for (let i = 0; i < 8; i++) {
        const s = document.createElement('span');
        s.className = 'spark';
        s.textContent = sparkEmojis[i % sparkEmojis.length];
        s.style.left = x + 'px';
        s.style.top = y + 'px';
        s.style.setProperty('--dx', (Math.random() * 160 - 80) + 'px');
        s.style.setProperty('--dy', (Math.random() * 160 - 80) + 'px');
        document.body.appendChild(s);
        setTimeout(function () { s.remove(); }, 900);
      }
    }
  </script>
